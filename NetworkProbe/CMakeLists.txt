cmake_minimum_required(VERSION 3.28)
project(ArizonaNetworkPortability LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 11)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT DEFINED RAKNET_SOURCE_DIR)
  message(FATAL_ERROR "Provide the locally fetched, pinned sampraknet source")
endif()
set(RAKNET_DLL OFF CACHE BOOL "Build static transport" FORCE)
add_subdirectory("${RAKNET_SOURCE_DIR}" transport)
add_executable(network-link-probe compile_link_probe.cpp)
target_link_libraries(network-link-probe PRIVATE raknet)
